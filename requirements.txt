# torch from PyPI on Linux pulls in the full CUDA stack (~3 GB), which the
# ~1 GB CPU-only Streamlit Cloud container cannot hold — the backend gets
# OOM-killed while loading its models. Pin PyTorch's CPU build there instead.
# macOS/Windows wheels are already CPU-only, so leave them unpinned.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.14.1+cpu ; sys_platform == "linux"
torch ; sys_platform != "linux"
